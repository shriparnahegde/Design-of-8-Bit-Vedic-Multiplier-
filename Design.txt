// 2x2 Vedic Multiplier (Base Module)
// ==========================================
module vedic_2x2 (
    input  [1:0] a,
    input  [1:0] b,
    output [3:0] p
);
    wire w1, w2, w3, carry1;

    assign p[0] = a[0] & b[0];
    assign w1   = a[1] & b[0];
    assign w2   = a[0] & b[1];
    assign w3   = a[1] & b[1];

    assign p[1]   = w1 ^ w2;
    assign carry1 = w1 & w2;

    assign p[2] = w3 ^ carry1;
    assign p[3] = w3 & carry1;
endmodule

// ==========================================
// 4x4 Vedic Multiplier
// ==========================================
module vedic_4x4 (
    input  [3:0] a,
    input  [3:0] b,
    output [7:0] p
);
    wire [3:0] q0, q1, q2, q3;

    // Instantiate four 2x2 multipliers
    vedic_2x2 M0 (.a(a[1:0]), .b(b[1:0]), .p(q0));
    vedic_2x2 M1 (.a(a[3:2]), .b(b[1:0]), .p(q1));
    vedic_2x2 M2 (.a(a[1:0]), .b(b[3:2]), .p(q2));
    vedic_2x2 M3 (.a(a[3:2]), .b(b[3:2]), .p(q3));

    // Optimized Addition Tree to reduce carry delay
    wire [4:0] mid_sum;
    wire [2:0] sum2;
    wire [3:0] sum3;

    assign mid_sum = q1 + q2;
    assign sum2    = q0[3:2] + mid_sum[1:0];
    assign sum3    = q3 + mid_sum[4:2] + sum2[2];

    // Final Product Mapping
    assign p[1:0] = q0[1:0];
    assign p[3:2] = sum2[1:0];
    assign p[7:4] = sum3;
endmodule

// ==========================================
// 8x8 Vedic Multiplier (Top Module)
// ==========================================
module vedic_8x8 (
    input  [7:0] a,
    input  [7:0] b,
    output [15:0] p
);
    wire [7:0] q0, q1, q2, q3;

    // Instantiate four 4x4 multipliers
    vedic_4x4 M0 (.a(a[3:0]), .b(b[3:0]), .p(q0));
    vedic_4x4 M1 (.a(a[7:4]), .b(b[3:0]), .p(q1));
    vedic_4x4 M2 (.a(a[3:0]), .b(b[7:4]), .p(q2));
    vedic_4x4 M3 (.a(a[7:4]), .b(b[7:4]), .p(q3));

    // Optimized Addition Tree for Partial Products
    wire [8:0] mid_sum;
    wire [4:0] sum2;
    wire [7:0] sum3;

    assign mid_sum = q1 + q2; 
    assign sum2    = q0[7:4] + mid_sum[3:0];
    assign sum3    = q3 + mid_sum[8:4] + sum2[4];

    // Final Product Mapping
    assign p[3:0]  = q0[3:0];
    assign p[7:4]  = sum2[3:0];
    assign p[15:8] = sum3;
endmodule
